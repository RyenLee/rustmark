### 1.2 两个可独立成立的漏洞

1. **cmd 二次解析（CWE-78）**：`std::process::Command::args` 在 Windows 拼接命令行时，对不含空格/空串的参数**不加引号**（`https://a.com&calc.exe` 无空格）。`cmd /C` 收到整行后把 `&`、`|`、`>`、`^`、`%VAR%` 当 cmd 语法，执行额外命令。
2. **任意本地程序执行**：`is_safe_to_open` 的 `:71` 分支把「任何不含 `://` 的字符串」判为本地目标并返回 `true`。`[x](C:\Windows\System32\calc.exe)`、`[x](./evil.exe)`、`[x](//host/share)`、`[x](\\host\share)` 全部通过，随后 `cmd /C start` 执行 exe / 触发 SMB 认证（UNC 凭据外泄）。

两者**都不需要 `cmd` 之外的额外条件**，前提仅是「用户点击文档里的链接」。当前 `is_safe_to_open` 只拦 `javascript:` / `vbscript:` / `data:` 三个小写前缀（`ports/system.rs:63-69`），且协议白名单比较大小写敏感（`HTTPS://` 被误拒，属可用性问题）。

### 1.3 目标设计

分两层，**先去掉 shell，再收紧目标策略**（顺序不能反：只收紧策略而不去 shell，`&` 仍会在合法 URL 的查询串里触发；只去 shell 而不收紧策略，本地 exe 仍会被执行）。

```
                    ┌──────────────────────────────┐
target ──► classify_open_target（ports，纯逻辑，零 IO）
                    └───────┬───────────┬──────────┘
                            │Remote     │LocalDocument
                            │           │
                            ▼           ▼
                     launch（无 shell）    launch（无 shell）
                            │RevealOnly │Reject
                            ▼           ▼
                   explorer /select   Err(InvalidInput)
```

- **端口层（`rustmark-ports/src/system.rs`）**新增纯函数分类，保留 `is_safe_to_open` 作为兼容门面（`Reject` → `false`）。
- **实现层（`rustmark-infra/src/infrastructure/system.rs`）**按分类选择动作，**任何分支都不经 cmd.exe**。
- **Windows 启动方式二选一**：
  - **方案 A（推荐，S）**：`rundll32.exe url.dll,FileProtocolHandler <target>`。参数经 `Command::arg` 直达 argv，无 shell；零新依赖、零 `unsafe`。
  - **方案 B（M，结构更干净）**：`ShellExecuteW` FFI。一次 `unsafe` FFI 调用、不起子进程；项目当前 `unsafe` 为零，需接受并隔离这一处。
- **本地目标永不执行**：可打开扩展名（文本/位图）才交关联程序；其余本地目标（含 `.exe`/`.html`/`.pdf`）改为「在文件管理器中定位」，绝不交给壳执行。
- **UNC / 协议相对 URL 一律拒绝**：`//host/share`、`\\host\share`、`//evil.com` 在 Windows 会解析成 SMB 网络路径或外部主机。

#### 1.4.1 `crates/rustmark-ports/src/system.rs`：新增分类（保留旧门面）

在 `pub trait ExternalOpener` 之后、`is_safe_to_open` 之前插入：

```rust
/// 外链打开目标的分类结果（纯逻辑，零 IO，可被任意层依赖）。
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum OpenTarget {
    /// 文档内锚点，调用方自行跳转（打开器应视为 no-op）。
    Anchor,
    /// http / https / mailto / tel，交系统协议处理器。
    Remote,
    /// 本地文本 / 位图，可交系统关联程序打开（不会执行代码）。
    LocalDocument,
    /// 本地但不该执行（可执行文件 / 归档 / HTML / PDF 等）：
    /// 仅允许「在文件管理器中定位」。
    RevealOnly,
    /// 明确不安全（XSS scheme、UNC、协议相对、控制字符、编码绕过、空串）。
    Reject,
}

/// 可安全交给关联程序打开的本地扩展名（**不含**可执行 / 脚本 / HTML / PDF）。
const OPENABLE_EXTENSIONS: &[&str] = &[
    "md", "markdown", "txt", "png", "jpg", "jpeg", "gif", "webp", "bmp",
];

/// 识别形如 `scheme:` 的前缀；Windows 盘符 `C:` 不算 scheme。
fn scheme_of(s: &str) -> Option<&str> {
    let colon = s.find(':')?;
    let scheme = &s[..colon];
    if scheme.is_empty()
        || !scheme
            .chars()
            .all(|c| c.is_ascii_alphanumeric() || c == '+' || c == '-' || c == '.')
    {
        return None;
    }
    // 单字母且后接 `/` 或 `\`：Windows 盘符（`C:\x` / `C:/x`）。
    if scheme.len() == 1 {
        return None;
    }
    Some(scheme)
}

/// 把本地串的扩展名取出来（小写；`file://` 先剥前缀）。
fn local_extension(target: &str) -> &str {
    let path = target.strip_prefix("file://").unwrap_or(target);
    path.rsplit('.').next().unwrap_or("")
}

/// 该目标应如何打开（纯函数，单测唯一入口）。
pub fn classify_open_target(raw: &str) -> OpenTarget {
    let t = raw.trim();
    if t.is_empty() {
        return OpenTarget::Reject;
    }
    // 控制字符（含 NUL）：命令行 / 文件名都不该出现。
    if t.chars().any(|c| c.is_control()) {
        return OpenTarget::Reject;
    }
    if t.starts_with('#') {
        return OpenTarget::Anchor;
    }
    // 协议相对 `//host` 与 UNC `\\host`：Windows 下可触发 SMB 认证外泄。
    if t.starts_with("//") || t.starts_with(r"\\") {
        return OpenTarget::Reject;
    }
    // XSS 向量（显式拦截，防 `://` 分支漏过）。
    let lower = t.to_ascii_lowercase();
    if lower.starts_with("javascript:") || lower.starts_with("vbscript:") || lower.starts_with("data:")
    {
        return OpenTarget::Reject;
    }

    match scheme_of(t) {
        // 绝对协议：仅白名单（大小写不敏感）。
        Some(scheme) => match scheme.to_ascii_lowercase().as_str() {
            "http" | "https" | "mailto" | "tel" => OpenTarget::Remote,
            "file" => classify_local(t),
            _ => OpenTarget::Reject,
        },
        // 无 scheme：相对 / 绝对本地路径。
        None => classify_local(t),
    }
}

fn classify_local(target: &str) -> OpenTarget {
    let path = target.strip_prefix("file://").unwrap_or(target);
    // 本地路径里的百分号 / 查询串 / 锚点：`x%2Eexe` 这类编码可绕过扩展名判定。
    if path.contains('%') || path.contains('?') || path.contains('#') {
        return OpenTarget::Reject;
    }
    let ext = local_extension(target).to_ascii_lowercase();
    if OPENABLE_EXTENSIONS.contains(&ext.as_str()) {
        OpenTarget::LocalDocument
    } else {
        // 可执行 / 脚本 / 归档 / HTML / PDF / 无扩展名：只定位，不执行。
        OpenTarget::RevealOnly
    }
}
```

#### 1.4.2 `crates/rustmark-infra/src/infrastructure/system.rs`：去掉 shell

```rust
use rustmark_ports::system::{ExternalOpener, OpenTarget, classify_open_target};

impl ExternalOpener for SystemExternalOpener {
    fn open(&self, target: &str) -> std::io::Result<()> {
        match classify_open_target(target) {
            // 锚点由表现层处理；走到这里视为 no-op，不惊动系统。
            OpenTarget::Anchor => Ok(()),
            OpenTarget::Reject => Err(std::io::Error::new(
                std::io::ErrorKind::InvalidInput,
                "拒绝打开：该链接目标属于不安全类型（协议 / UNC / 可执行文件）",
            )),
            OpenTarget::LocalDocument => launch(target),
            OpenTarget::RevealOnly => reveal(target),
        }
    }
}

/// 启动 URL / 可打开文档。**任何平台分支都不得经过 shell**。
fn launch(target: &str) -> std::io::Result<()> {
    #[cfg(target_os = "windows")]
    {
        launch_command(target).spawn()?;
    }
    #[cfg(target_os = "macos")]
    {
        std::process::Command::new("open").arg(target).spawn()?;
    }
    #[cfg(target_os = "linux")]
    {
        std::process::Command::new("xdg-open").arg(target).spawn()?;
    }
    #[cfg(not(any(target_os = "windows", target_os = "macos", target_os = "linux")))]
    {
        let _ = target;
        return Err(std::io::Error::new(
            std::io::ErrorKind::Unsupported,
            "unsupported platform for ExternalOpener",
        ));
    }
    Ok(())
}

/// 只在文件管理器中定位，不执行目标。
fn reveal(target: &str) -> std::io::Result<()> {
    #[cfg(target_os = "windows")]
    {
        // 注意：explorer.exe 即使成功也常返回非 0；这里只 spawn，不看退出码。
        let path = local_path_of(target);
        std::process::Command::new("explorer.exe")
            .arg(format!("/select,{path}"))
            .spawn()?;
    }
    #[cfg(target_os = "macos")]
    {
        std::process::Command::new("open").arg("-R").arg(local_path_of(target)).spawn()?;
    }
    #[cfg(target_os = "linux")]
    {
        // 无统一「定位」协议：退化为打开父目录。
        let path = local_path_of(target);
        let parent = std::path::Path::new(&path)
            .parent()
            .map(|p| p.to_path_buf())
            .unwrap_or_else(|| std::path::PathBuf::from("."));
        std::process::Command::new("xdg-open").arg(parent).spawn()?;
    }
    #[cfg(not(any(target_os = "windows", target_os = "macos", target_os = "linux")))]
    {
        let _ = target;
        return Err(std::io::Error::new(
            std::io::ErrorKind::Unsupported,
            "unsupported platform for reveal",
        ));
    }
    Ok(())
}

/// Windows 启动命令的纯构造（与 `open()` 分离，便于单测断言参数）。
#[cfg(target_os = "windows")]
pub fn launch_command(target: &str) -> std::process::Command {
    let mut command = std::process::Command::new("rundll32.exe");
    // argv 直达，不经 cmd.exe；`&`、`|` 等不再有语法含义。
    command.args(["url.dll,FileProtocolHandler", target]);
    command
}

/// `file:///C:/x` → `C:/x`（Windows 去掉 `file://` 后的前导 `/`）。
fn local_path_of(target: &str) -> String {
    let p = target.strip_prefix("file://").unwrap_or(target);
    match p.strip_prefix('/') {
        Some(rest) if rest.len() > 1 && rest.as_bytes().get(1) == Some(&b':') => rest.to_string(),
        _ => p.to_string(),
    }
}
```

#### 1.4.3 可选：方案 B —— `ShellExecuteW`（不推荐在本轮引入，除非接受一处 `unsafe`）

`windows-sys 0.61.2` 已在 `Cargo.lock` 中（由 eframe / wgpu 传递引入），新增直接依赖不会引入新版本。已核对源码确认所需 feature 与签名：

- 函数：`windows_sys::Win32::UI::Shell::ShellExecuteW`
- 依赖 feature：`Win32_Foundation` + `Win32_UI_Shell` + `Win32_UI_WindowsAndMessaging`
  （签名引用 `Foundation::HWND`、`Foundation::HINSTANCE`、`WindowsAndMessaging::SHOW_WINDOW_CMD`）
- `HINSTANCE = *mut core::ffi::c_void`；`SW_SHOWNORMAL: SHOW_WINDOW_CMD = 1i32`；返回 `<= 32` 视为失败。

`crates/rustmark-infra/Cargo.toml` 末尾追加（**这是唯一需要改清单的路径**）：

```toml
[target.'cfg(target_os = "windows")'.dependencies]
windows-sys = { version = "0.61", features = [
    "Win32_Foundation",
    "Win32_UI_Shell",
    "Win32_UI_WindowsAndMessaging",
] }
```

实现片段（仅方案 B，隔离在一个函数内）：

```rust
#[cfg(target_os = "windows")]
fn launch(target: &str) -> std::io::Result<()> {
    use windows_sys::Win32::Foundation::HINSTANCE;
    use windows_sys::Win32::UI::Shell::ShellExecuteW;
    use windows_sys::Win32::UI::WindowsAndMessaging::SW_SHOWNORMAL;

    let wide: Vec<u16> = target.encode_utf16().chain(std::iter::once(0)).collect();
    // SAFETY: 空窗口句柄 + 空操作动词 = 默认动词；`wide` 以 NUL 结尾且在本调用期间存活。
    let rc: HINSTANCE = unsafe {
        ShellExecuteW(
            std::ptr::null_mut(),
            std::ptr::null(),
            wide.as_ptr(),
            std::ptr::null(),
            std::ptr::null(),
            SW_SHOWNORMAL,
        )
    };
    if (rc as isize) <= 32 {
        return Err(std::io::Error::other(format!(
            "ShellExecuteW 失败（code {}）",
            rc as isize
        )));
    }
    Ok(())
}
```

> `std::io::Error::other` 在 Rust 1.99 可用（待核对，若不可用改 `io::Error::new(ErrorKind::Other, …)`）。
> 方案 B 仍需 1.3 的分类策略；它只消除 shell，不阻止本地 exe 被执行。

### 1.5 策略细节

| 输入形态 | 分类 | 行为 | 理由 |
|---|---|---|---|
| `#anchor` | `Anchor` | no-op（表现层先跳转） | 文档内导航 |
| `http://` / `https://` / `mailto:` / `tel:`（大小写不敏感） | `Remote` | launch | 白名单协议 |
| `HTTPS://example.com` | `Remote` | launch | 修复原大小写敏感缺陷 |
| `//host/share`、`//evil.com` | `Reject` | 拒绝 | 协议相对 / Windows 解析为 UNC |
| `\\host\share` | `Reject` | 拒绝 | UNC → SMB 认证外泄 |
| `javascript:` / `vbscript:` / `data:`（任意大小写） | `Reject` | 拒绝 | XSS 向量 |
| 其他未知 scheme（`ftp:`、`filex:`…） | `Reject` | 拒绝 | 默认拒绝 |
| `file:///C:/x.png` / `./foo.png` / `relative/path.md` | `LocalDocument` | launch | 文本 / 位图，不执行 |
| `C:\x\evil.exe`、`./evil.bat`、`x.ps1`、`x.lnk` | `RevealOnly` | 文件管理器定位 | **绝不执行** |
| `./x.html`、`./x.pdf`、`./x.svg` | `RevealOnly` | 定位 | 浏览器 / PDF 阅读器可执行脚本，保守处理 |
| `./x.docx`、无扩展名、未知扩展名 | `RevealOnly` | 定位 | 不交壳执行 |
| 本地串含 `%` / `?` / `#`（如 `x%2Eexe`） | `Reject` | 拒绝 | 百分号编码可绕过扩展名判断 |
| 含控制字符（`\u{0}`、`\n`、`\r`） | `Reject` | 拒绝 | 命令行 / 文件名污染 |
| 空串 / 纯空白 | `Reject` | 拒绝 | 无目标 |
| 合法 URL 查询串含 `&`（`https://a.com?a=1&b=2`） | `Remote` | launch | **不再需要字符过滤**——shell 已移除 |

关于 cmd 元字符：**不采用全局字符拒绝**。`&`、`%20` 在合法 URL 里完全正常；根因是 `cmd /C` 的二次解析，去掉 shell 后即消失。若团队坚持双保险，可在 `Remote` 分支额外拒绝 `& | ^ < > " \``，但会误伤合法查询串，不建议。

百分号编码：`Remote` 分支允许（`%20` 合法）；本地分支一律拒绝（避免 `%2E` → `.` 的扩展名绕过）。

大小写：scheme 判定统一 `to_ascii_lowercase`；路径扩展名同样小写化。

空 target：`Reject`（原实现会 spawn 一个 `cmd /C start "" ""`，本身就是缺陷；现有单测 `open_takes_a_borrowed_target` 调 `open("")`，修复后应返回 `Err` 而非起进程）。

本地文件操作改为「定位」还是「拒绝」：本方案选**定位**（`RevealOnly`），把「拒绝」限定在真正的危险输入（XSS / UNC / 编码绕过 / 控制字符）。若产品不接受「定位」的 UX，可把 `RevealOnly` 降级为 `Reject`，一行改动。

### 2.2 目标设计

1. **主防线**：`build_print_html` 在拼装打印文档前，对「正文 + TOC」整体调用 `sanitize_html`（与 HTML 导出同一门）。清洗作用于 **fragment**（pulldown 输出），不作用于 `wrap_document` 生成的骨架——否则 `sanitize_html` 会把 `<meta charset>`、`<style>` 一并剥掉（`style`/`meta` 在 `DROP_TAGS`，见 `html_sanitize/policy.rs:51-54`），打印变成无样式裸文档。
2. **第二道防线**：无头 Chromium 追加 `--blink-settings=scriptEnabled=false`。**但仅在文档不需要公式/图表时**加——含 KaTeX/Mermaid 时禁用脚本会让公式/图表退回源码形态（这是当前 PDF 里唯一依赖脚本的渲染路径）。
3. **注释纠偏**：`build_printable_document` 与 `build_print_html` 的文档注释必须改写，删除「不会在浏览器里打开 / 无脚本注入面」的错误前提。
4. **兜底路径同样受益**：`shell_print_to`（`dispatch.rs:219`）交给 MSHTML `printto` 也会执行脚本；清洗后两条路径都安全，无需为兜底单独改。

```rust
    /// 生成打印就绪的 HTML（含内联样式，供打印使用）。
    ///
    /// # 为什么这里**必须**清洗
    ///
    /// 产物会被无头 Chromium 以 `file:///` 打开并渲染成 PDF——那是一个真实的
    /// 浏览器引擎，文档里的 `<script>` / 事件属性会执行。这与 HTML 导出
    /// （`PulldownHtmlExporter::build_document`）面对的是同一类不可信输入，
    /// 因此复用同一套白名单清洗，且清洗只作用于**正文 + TOC 片段**，
    /// 不碰 `wrap_document` 生成的 `<meta>` / `<style>` 骨架。
    ///
    /// 公开仅为让单测能直接断言中间产物（同 `PulldownHtmlExporter::build_document`）。
    pub fn build_print_html(content: &str, job: &ExportJob) -> String {
        let toc: Vec<TocEntry> = collect_toc(content);
        let _ = TocBuilder::verify(&toc);
        let title = job.destination().file_name().to_string();
        let body = render_fragment(content);
        let with_toc = if job.options().include_toc() && job.needs_toc() {
            format!("{body}\n{}", render_toc(&toc))
        } else {
            body
        };
        // 不可信来源（pulldown 原样透传的内嵌 HTML + TOC）先过白名单清洗。
        let safe_body = sanitize_html(&with_toc);
        let page_style = format!(
            "@page{{size:{};margin:18mm}}",
            match job.options().page_size() {
                PageSize::A4 => "A4",
                PageSize::Letter => "letter",
            }
        );
        super::pulldown_html_exporter::build_printable_document(&title, &safe_body, &page_style)
    }
```

```rust
#[cfg(target_os = "windows")]
fn headless_print(browser: &Path, html: &Path, pdf: &Path) -> Result<u64, String> {
    let output = headless_print_command(browser, html, pdf)
        .output()
        .map_err(|e| format!("调起 {} 失败：{e}", browser.display()))?;

    let size = std::fs::metadata(pdf).map(|meta| meta.len()).map_err(|e| {
        format!(
            "{} 未产出 PDF（{}）：{}",
            browser.display(),
            e,
            String::from_utf8_lossy(&output.stderr).trim()
        )
    })?;
    Ok(size)
}

/// 构造无头打印命令（纯构造，便于单测断言参数而**不真起进程**）。
///
/// 第二道防线：文档不含公式 / 图表（无需注入 CDN 脚本）时禁用脚本执行。
/// 含 KaTeX / Mermaid 时**不能**禁用——它们正是靠注入脚本渲染进 PDF 的；
/// 此时第一道防线（`sanitize_html` 已剥除用户脚本）承担全部责任。
#[cfg(target_os = "windows")]
pub fn headless_print_command(browser: &Path, html: &Path, pdf: &Path) -> Command {
    let mut command = Command::new(browser);
    command.args([
        "--headless=new",
        "--disable-gpu",
        "--no-first-run",
        "--no-default-browser-check",
        "--no-pdf-header-footer",
    ]);
    let needs_scripts = std::fs::read_to_string(html)
        .map(|s| s.contains("class=\"math") || s.contains("language-mermaid"))
        .unwrap_or(false);
    if !needs_scripts {
        command.arg("--blink-settings=scriptEnabled=false");
    }
    command
        .arg(format!("--print-to-pdf={}", pdf.display()))
        .arg(file_url(html));
    command
}
```
